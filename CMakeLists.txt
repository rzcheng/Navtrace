cmake_minimum_required(VERSION 3.24)
project(NavTrace VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NAVTRACE_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
include(FetchContent)

find_package(Eigen3 3.4 QUIET NO_MODULE)
if(NOT Eigen3_FOUND)
  set(eigen_archive "https://gitlab.com/libeigen/eigen/-/archive/3.4.0/eigen-3.4.0.tar.gz")
  if(EXISTS "${CMAKE_SOURCE_DIR}/.deps/eigen-3.4.0.tar.gz")
    set(eigen_archive "${CMAKE_SOURCE_DIR}/.deps/eigen-3.4.0.tar.gz")
  endif()
  # Only use the headers; do not configure Eigen's own test suite.
  FetchContent_Declare(eigen URL "${eigen_archive}"
    URL_HASH SHA256=8586084f71f9bde545ee7fa6d00288b264a2b7ac3607b974e54d13e7162c1c72
    SOURCE_SUBDIR navtrace-unused-subdir)
  FetchContent_MakeAvailable(eigen)
  add_library(navtrace_eigen INTERFACE)
  target_include_directories(navtrace_eigen SYSTEM INTERFACE "${eigen_SOURCE_DIR}")
  add_library(Eigen3::Eigen ALIAS navtrace_eigen)
  set(Eigen3_VERSION 3.4.0)
endif()

add_library(navtrace src/ekf.cpp src/replay.cpp src/io.cpp)
target_include_directories(navtrace PUBLIC include)
target_link_libraries(navtrace PUBLIC Eigen3::Eigen)
add_executable(navtrace_replay apps/navtrace_replay.cpp)
target_link_libraries(navtrace_replay PRIVATE navtrace)

function(navtrace_checks target)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    if(NAVTRACE_SANITIZE)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  elseif(NAVTRACE_SANITIZE)
    message(FATAL_ERROR "Sanitizers require Clang or GCC")
  endif()
endfunction()
navtrace_checks(navtrace)
navtrace_checks(navtrace_replay)

include(CTest)
if(BUILD_TESTING)
  find_package(GTest 1.17 QUIET CONFIG)
  if(NOT GTest_FOUND)
    set(gtest_archive "https://github.com/google/googletest/archive/refs/tags/v1.17.0.tar.gz")
    if(EXISTS "${CMAKE_SOURCE_DIR}/.deps/googletest-v1.17.0.tar.gz")
      set(gtest_archive "${CMAKE_SOURCE_DIR}/.deps/googletest-v1.17.0.tar.gz")
    endif()
    FetchContent_Declare(googletest URL "${gtest_archive}"
      URL_HASH SHA256=65fab701d9829d38cb77c14acdc431d2108bfdbf8979e40eb8ae567edf10b27c)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
    set(GTest_VERSION 1.17.0)
  endif()
  add_executable(navtrace_tests tests/test_ekf.cpp tests/test_replay.cpp)
  target_link_libraries(navtrace_tests PRIVATE navtrace GTest::gtest_main)
  navtrace_checks(navtrace_tests)
  include(GoogleTest)
  gtest_discover_tests(navtrace_tests)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/build_info.txt" CONTENT
"project=NavTrace 0.1.0\ncompiler=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\ncompiler_path=${CMAKE_CXX_COMPILER}\nsystem=${CMAKE_SYSTEM_NAME} ${CMAKE_SYSTEM_PROCESSOR}\nbuild_type=$<CONFIG>\ncxx_standard=17\neigen=${Eigen3_VERSION}\ngoogletest=${GTest_VERSION}\nsanitizers=${NAVTRACE_SANITIZE}\n")
